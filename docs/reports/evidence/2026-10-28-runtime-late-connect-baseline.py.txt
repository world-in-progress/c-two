"""Deterministic regressions for SDK session retirement (baseline is red)."""
from __future__ import annotations

import json
import os
from queue import Queue
from threading import Event, Lock, Thread, current_thread

import pytest

import c_two as cc
from c_two.crm.contract import CRMContract
from c_two.transport import registry
from c_two.transport.client.proxy import CRMProxy


@cc.crm(namespace='test.runtime-retirement', version='0.1.0')
class Echo:
    def echo(self, value: str) -> str: ...


class EchoResource:
    def echo(self, value: str) -> str:
        return value


@pytest.mark.parametrize('hosted', [False, True], ids=['hostless', 'hosted'])
@pytest.mark.parametrize('expire_at_finish', [False, True], ids=['live-budget', 'expired-budget'])
def test_shutdown_cannot_publish_a_connection_from_its_retired_session(
    tmp_path, monkeypatch, hosted, expire_at_finish,
):
    for key in tuple(os.environ):
        if key.startswith('C2_'):
            monkeypatch.delenv(key)
    monkeypatch.setenv('C2_ENV_FILE', '')
    server, client = cc.Runtime(), cc.Runtime()
    if os.name != 'nt':
        for runtime in (server, client):
            runtime.set_local_endpoint(root=str(tmp_path))
    old_session = client._runtime_session  # Keep ownership explicit through cleanup.
    events = client.connection_events()
    captured, resume = Event(), Event()
    reached_finish, resume_finish = Event(), Event()
    attempts, observations = Queue(), Queue()
    result = Queue()
    original_args = CRMContract.native_args
    original_call = registry._connect_native_call
    original_ipc = CRMProxy.ipc
    observation = None
    late_proxy = None
    evidence = {'hosted': hosted, 'expire_at_finish': expire_at_finish}

    def native_args(contract):
        if current_thread() is worker:
            # Runtime._connect has captured its session, but has not entered
            # NativeSession.acquire_ipc_client. No native result is faked.
            captured.set()
            assert resume.wait(5), 'test did not release the captured connect'
        return original_args(contract)

    def ipc(*args, **kwargs):
        proxy = original_ipc(*args, **kwargs)
        if current_thread() is worker:
            observations.put(proxy.connection_observation())
        return proxy

    def native_call(operation, *args, **kwargs):
        if expire_at_finish and current_thread() is worker and args == ('python_connect_finish',):
            attempts.put(operation.__self__)
            reached_finish.set()
            assert resume_finish.wait(5), 'test did not release final publication'
        return original_call(operation, *args, **kwargs)

    def connect():
        try:
            result.put(client.connect(
                Echo, name='echo', address=server.server_address(), timeout=3,
            ))
        except BaseException as error:
            result.put(error)  # Keep the traceback and old owners alive too.
        finally:
            reached_finish.set()

    worker = Thread(target=connect, name='captured-runtime-connect')
    try:
        server.register(Echo, EchoResource(), name='echo')
        if hosted:
            client.start()
        monkeypatch.setattr(CRMContract, 'native_args', native_args)
        monkeypatch.setattr(CRMProxy, 'ipc', ipc)
        monkeypatch.setattr(registry, '_connect_native_call', native_call)
        worker.start()
        assert captured.wait(5), 'connect did not reach the native boundary'
        outcome = client.shutdown(timeout=2)
        evidence['shutdown_completed'] = outcome['completed']
        evidence['shutdown_ipc_clients_drained'] = outcome['ipc_clients_drained']
        assert outcome['completed'] and outcome['ipc_clients_drained']
        assert client._runtime_session is not old_session
        resume.set()
        if expire_at_finish:
            assert reached_finish.wait(5)
            if not attempts.empty():
                attempt = attempts.get_nowait()
                # Expire the original native budget on a real held lock;
                # no sleep, clock replacement, or fresh timeout is involved.
                blocked = Lock()
                blocked.acquire()
                try:
                    with pytest.raises(cc.error.CoreError):
                        attempt.acquire_lock(blocked, 'fixture_expire_original_budget')
                finally:
                    blocked.release()
            resume_finish.set()
        worker.join(5)
        assert not worker.is_alive()
        returned = result.get_nowait()
        evidence['connect_result'] = type(returned).__name__
        if not isinstance(returned, BaseException):
            late_proxy = returned
            evidence['late_call'] = late_proxy.echo('still-alive')
            cc.close(late_proxy)
        else:
            evidence['connect_error'] = str(returned)
        if not observations.empty():
            observation = observations.get_nowait()
            event = events.recv(timeout=1)
            while event is not None and event.side != 'client':
                event = events.recv(timeout=1)
            assert event.kind == 'opened' and event.side == 'client'
            assert event.connection_id == observation.connection_id
            evidence['closed_after_proxy_close'] = observation.snapshot() is not None
            repeated = old_session.shutdown(timeout_seconds=1)
            evidence['old_session_reobserve_completed'] = repeated['completed']
            evidence['old_session_reobserve_drained'] = repeated['ipc_clients_drained']
            evidence['closed_after_old_shutdown'] = observation.snapshot() is not None
    finally:
        resume.set()
        resume_finish.set()
        if worker.ident is not None:
            worker.join(5)
            assert not worker.is_alive()
        if late_proxy is not None:
            cc.close(late_proxy)
        # The real peer's close barrier also collects the baseline's late
        # hosted connection; old Host.shutdown otherwise just replays its journal.
        assert server.shutdown()['completed']
        assert old_session.shutdown()['completed']
        assert client.shutdown()['completed']
        events.close()
        if observation is not None:
            assert observation.snapshot() is not None
        remaining = sorted(path.name for path in tmp_path.iterdir())
        assert set(remaining) <= {'.gate', '.gate.marker'}
        evidence['endpoint_remaining'] = remaining
        print('LATE_CONNECT_BASELINE ' + json.dumps(evidence, sort_keys=True))

    assert isinstance(returned, (RuntimeError, cc.error.CallDeadlineExceeded)), (
        'shutdown reported drained, but the captured old session published '
        f'a new usable connection: {evidence}'
    )
    assert evidence.get('closed_after_proxy_close', True), (
        'connect failed within its original budget, but its physical connection '
        f'escaped the completed shutdown: {evidence}'
    )
